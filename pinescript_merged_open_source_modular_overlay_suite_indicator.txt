//@function         Returns the centerline value for the `type` oscillator.
//@param type       (simple string) The type of oscillator (uses constants that must be defined earlier in the script).
//@returns          (series float) The centerline's value.
oscCenter(simple string type) =>
    type == OS01 or type == OS05 or type == OS06 or type == OS09 or type == OS12 or type == OS14 ? 50. :
    type == OS11 ? -50. :
    0.

//@function     Determines a "plot_style" to be used from a user's input.
//@param state  (input string) The user selection string of his line style choice (depends on the `STL1`, `STL2` and `STL3` string constants).
//@returns      (plot_style) The `style` named argument required in `plot()`.
lineStyleFromUserInput(userSelection) =>
    userSelection == STL1 ? plot.style_line :
    userSelection == STL2 ? plot.style_circles :
    userSelection == STL3 ? plot.style_cross :
    plot.style_line

ow_transitionTo(series bool state) =>
    not state[1] and state

ow_macdHist(series float src) =>
    [ow_macdLine, ow_signalLine, ow_histLine] = ta.macd(src, 12, 26, 9)
    ow_histLine

ow_kvo(simple int length) =>
    [ow_kvoLine, ow_kvoSignal] = TvTa.kvo(length, length * 2, length / 2)
    ow_kvoLine

ow_aroonDiff(simple int length) =>
    [ow_up, ow_dn] = TvTa.aroon(length)
    ow_up - ow_dn

ow_dmiDiff(simple int length) =>
    [ow_up, ow_dn, ow_adx] = ta.dmi(length, length)
    ow_up - ow_dn

ow_trixLine(series float src, simple int length) =>
    [ow_trix, ow_signal, ow_hist] = TvTa.trix(src, length, int(length * 0.66))
    ow_trix

ow_oscCalc(simple string type, series float src, simple int length) =>
    float result =
         type == OS00 ? na :
         type == OS01 ? ta.rsi(src, length) :
         type == OS02 ? ow_macdHist(src) :
         type == OS03 ? ta.cci(src, length) :
         type == OS04 ? ta.cmo(src, length) :
         type == OS05 ? ta.stoch(src, high, low, length) :
         type == OS06 ? ta.sma(ta.stoch(ta.rsi(src, length), ta.rsi(src, length), ta.rsi(src, length), length), 3) :
         type == OS09 ? ta.mfi(hlc3, length) :
         type == OS10 ? ta.tsi(src, length, length * 2) :
         type == OS11 ? ta.wpr(length) :
         type == OS12 ? TvTa.uo(length / 2, length, length * 2) :
         type == OS13 ? ow_kvo(length) :
         type == OS14 ? TvTa.stc(src, length, length * 2, length / 2, 3, 3) :
         type == OS15 ? TvTa.vzo(length) :
         type == OS16 ? TvTa.pzo(length) :
         type == OS17 ? TvTa.szo(src, length) :
         type == OS18 ? TvTa.wpo(length) :
         type == OS19 ? LucfTa.buoyancy(volume, length, 1000) :
         type == OS20 ? TvTa.coppock(src, length, length / 2, length / 2) :
         type == OS21 ? ow_aroonDiff(length) :
         type == OS22 ? ow_dmiDiff(length) :
         type == OS23 ? ow_trixLine(src, length) :
         type == OS24 ? TvTa.eom(length) :
         type == OS25 ? TvTa.ft(src, length) :
         type == OS26 ? LucfTa.sott() :
         type == OS27 ? TvTa.ht(src) :
         type == OS28 ? ta.iii :
         type == OS52 ? ta.tr * math.sign(ta.change(close)) :
         type == OS54 ? ta.mom(src, length) :
         type == OS55 ? ta.roc(src, length) :
         type == OS90 ? LucfTa.efficientWork(length) :
         na
    result

ow_oscWeight(series float osc, simple float center = 0.0) =>
    float ow_normalizedOsc = osc - center
    float ow_historicalMax = math.abs(ta.max(ow_normalizedOsc))
    float ow_historicalMin = math.abs(ta.min(ow_normalizedOsc))
    osc > center ? ow_normalizedOsc / ow_historicalMax : osc < center ? -math.abs(ow_normalizedOsc / ow_historicalMin) : 0.

ow_checkForNoVolume(simple string type) =>
    bool ow_oscUsesVolume = type == OS09 or type == OS15 or type == OS19 or type == OS24
    if barstate.islast and ta.cum(nz(volume)) == 0 and ow_oscUsesVolume
        runtime.error("No volume is provided by the data vendor.")
